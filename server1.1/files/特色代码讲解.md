## 一、串口一问一答协议（防丢包绘图指令）

**来源：`tft_display.py` — TFTDisplay.cmd()**

```python
def cmd(self, text) -> bool:
    """发送一条绘图指令并等待 OK 应答，失败自动重试一次。

    输入：text 指令字符串（不带换行），如 "FRECT 12 24 104 96 0"
    输出：True 固件应答 OK；串口异常时置 _ok=False 并返回 False。
    """
    if self._ser is None:
        return False                      # 未连接串口，直接放弃
    packet = (text.strip() + "\n").encode("ascii")
    try:
        with self._lock:                  # 加锁：动画线程与主线程共用一个串口
            for _ in range(2):            # 失败自动重试一次
                self._ser.write(packet)
                resp = self._ser.readline().decode("ascii", "ignore").strip()
                if resp == "OK":          # 下位机执行完才回 OK —— 保证不丢帧
                    return True
        self._ok = False                  # 连续两次无应答：判定屏幕掉线，暂停绘制
        return False
    except Exception:
        self._ok = False                  # 串口异常（拔线等）同样降级
        return False
```

**讲解稿：**

> 大家画图时一般是一条指令丢过去就完事，但串口是流式的，发太快指令会互相"粘"在一起，屏幕上就会出现花屏。所以我们设计了一问一答协议：上位机每发一条绘图指令就阻塞等下位机回 `OK`，没收到就重发一次，连续两次没响应就判定屏幕掉线、自动暂停动画——主打靶流程完全不受影响，屏幕重新插上后线程里每秒探测一次，收到 `OK` 立刻恢复显示。整个动画系统跑在后台线程，和视觉主循环零耦合。

## 二、TFT 状态屏动画引擎（状态机 + 分层绘制）

**来源：`tft_display.py` — _render() 与雷达余辉动画**

```python
def _render(self):
    """画一帧：状态变化先重画静态层，再只更新动画区域。

    分两层的好处：静态层（标题栏、雷达底座）只在切状态时画一次，
    每帧只重画会动的那一小块区域 —— 大幅减少串口指令数，10fps 无压力。
    """
    state, info = self._state, self._info
    if state != self._static_state:       # 只有切状态才重画整屏静态层
        self._draw_static(state)
        self._static_state = state
        self._tick = 0                    # 动画帧计数归零

    # 按当前状态分发到对应的动画函数
    if state == "SEARCH":
        self._anim_search()
    elif state == "AIM":
        self._anim_aim(info)
    elif state == "MOVE":
        self._anim_move(info)
    elif state == "FIRE":
        self._anim_fire()
    elif state == "IDLE":
        self._anim_idle()
    elif state == "DONE":
        self._anim_done()
    elif state == "RELOAD":
        self._anim_reload()
    self._draw_info(info)                 # 底部信息栏：靶号/进度/偏差
    self._tick += 1


def _anim_search(self):
    """搜索状态：雷达余辉扫描（主线 + 两道渐暗拖尾）+ 目标闪点。

    亮点：模拟真实雷达的"余辉"效果 —— 扫描线扫过去之后，
    身后留下两道颜色逐级变暗的拖尾，视觉上非常有科技感。
    """
    r = 46                                # 雷达半径
    step = 15                             # 每帧转 15 度，约 1.5 秒转一圈
    # 第一步：把上一帧的三条扫描线全部擦掉（用背景色覆盖）
    for back in (1, 2, 3):
        ang_old = math.radians((self._tick - back) * step)
        self.cmd(f"LINE {CX} {CY} {CX + int(r * math.cos(ang_old))} "
                 f"{CY + int(r * math.sin(ang_old))} {BLACK}")
    # 第二步：画新扫描线 + 余辉（绿 → 暗绿 → 更暗，亮度逐级衰减）
    colors = (GREEN, DGREEN, DARK)
    for back, c in enumerate(colors):
        ang = math.radians((self._tick - back) * step)
        self.cmd(f"LINE {CX} {CY} {CX + int(r * math.cos(ang))} "
                 f"{CY + int(r * math.sin(ang))} {c}")
    # 目标闪点：扫描线扫到 40 度方向附近时，目标"被照亮"闪红
    blip_phase = self._tick % 24
    if blip_phase < 8:
        self.cmd(f"FCIRCLE {CX + 30} {CY - 24} 4 {RED}")
    else:
        self.cmd(f"FCIRCLE {CX + 30} {CY - 24} 4 {BLACK}")
```

**讲解稿：**

> 屏幕 UI 的核心是一个七状态动画引擎：待机、搜索、瞄准、运动、发射、换弹、完成，每个状态一套专属动画。技术上做了两个优化：第一是**分层绘制**——标题栏和雷达底座这些不动的东西只在切换状态时画一次，每帧只更新会动的小区域，串口指令量降了一个数量级；第二是**擦旧画新**，不用整屏清除，避免闪烁。大家现在看到的是搜索状态的雷达动画，扫描线带两道渐暗余辉，目标只在被扫过时闪亮——这套 UI 让操作手在两米外就能一眼读出机器人在干什么。

## 三、云台对准：脉冲步进 + 迟滞带 + 超限放行

**讲解稿：**

> 对准算法是整个项目踩坑最多、也是最有技术含量的部分，它解决了三个真实硬件问题：
>
> 第一，**惯性冲过头**。高速冲到瞄点附近一帧就冲出去，然后正反方向每帧互相抵消，云台看起来"冻住"了。解决方案是距离 120px 以内切换成**脉冲步进**：走 3 帧、停 2 帧等惯性完全停稳、再重新测量，像人手一样一步步蹭到瞄点。
>
> 第二，**边界抖动**。偏差在容差线两侧反复横跳，电机每帧被反向踢一次。我们加了**迟滞带**：停住之后只要偏差没出带就不反向踢电机，直接计入稳定帧。
>
> 第三，**永不收敛**。机械死区或光照原因偶尔就是凑不进容差带，流程不能永远卡住。所以微调计步，超过 12 次强制放行执行动作，日志里会记录当时的实际偏差，方便赛后校准。激光任务还单独把容差收紧到 ±7px，最后一号靶则把脉冲速度拉到 255 满速，实现快速定位——同一个算法，按靶位灵活配参。

## 四、全流程动作音效系统

**来源：`track_fire.py` — 音效常量表与 beep()**

```python
SOUND_ON = True         # 总开关：比赛现场需要静音时改成 False 即可

# 每个动作一组 (频率, 时长秒) 序列 —— 每个动作都有专属音效
SND_FOUND     = [(880, 0.06), (1175, 0.06)]   # 发现目标：两声上行"叮咚"
SND_ALIGNED   = [(660, 0.06), (990, 0.08)]    # 对准到位：低→高确认音
SND_FIRE      = [(1568, 0.05), (1047, 0.06)]  # 发射：高频"哔——啵"
SND_LASER_ON  = [(1319, 0.06)]                # 激光开：单音提示
SND_LASER_OFF = [(659, 0.06)]                 # 激光关：低音提示
SND_BTN_A     = [(988, 0.12)]                 # 收到换弹按键A：长鸣
SND_DONE      = [(523, 0.1), (659, 0.1), (784, 0.1), (1047, 0.18)]
                                              # 全部完成：do-mi-so-do 上行琶音

def beep(notes):
    """按 (频率, 时长) 序列驱动主板蜂鸣器。任何异常都静默吞掉——
    音效只是锦上添花，绝不能影响打靶主流程。"""
    if not SOUND_ON:
        return
    try:
        for freq, dur in notes:
            board.buzzer_tone(freq)      # 发 @BU 指令：固件 tone() 发声
            time.sleep(dur)
            board.buzzer_tone(0)         # 频率 0 = 静音，音与音之间断开
            time.sleep(0.03)
    except Exception:
        pass
```

**讲解稿：**

> 机器人在场上跑，操作手盯着屏幕看不到所有细节，所以我们给每个动作都配了专属蜂鸣音效：发现目标"叮咚"、对准到位一声确认、发射"哔啵"、换弹等待时循环播放 do-re-mi-fa、全部完成来一段上行琶音——闭着眼睛都能听出机器人打到哪一步了。实现上是个四十行的音效表驱动：一个 `beep()` 函数加一组频率常量，而且全部包在 try 里静默容错，音效系统炸了也绝不影响打靶。顶部一个开关，赛前检测时想静音一改就行。

## 五、固件端安全：舵机双重限位 + 蜂鸣命令

**来源：`firmware/guard_robot/guard_robot.ino`**

```cpp
// ---- 舵机物理安全范围：60~120 度（云台俯仰机构超出会顶死）----
#define SERVO_ANGLE_MIN 60
#define SERVO_ANGLE_MAX 120

// 上位机发 @S<角度> 设置舵机。入口先限一次……
if (strcmp(op, "S") == 0) {
    int a;
    sscanf(cmd, "S %d", &a);
    a = constrain(a, SERVO_ANGLE_MIN, SERVO_ANGLE_MAX);  // 第一重：命令入口
    moveServoTo(a);
    sendAck(true);
}

void moveServoTo(int angle) {
    // 第二重：执行函数内部再限一次 —— 双保险，
    // 即使以后新代码绕过入口直接调用，也不可能顶坏机械结构
    angle = constrain(angle, SERVO_ANGLE_MIN, SERVO_ANGLE_MAX);
    servo.write(angle);
}

// 上位机发 @BU<频率> 控制蜂鸣器（0 = 静音）。之前固件根本没实现
// 这条命令，上位机发什么都没反应 —— 我们补齐了它，音效系统才有硬件支撑
if (strcmp(op, "BU") == 0) {
    int f;
    sscanf(cmd, "BU %d", &f);
    if (f > 0) tone(BUZZER_PIN, f);     // 有频率：持续发声
    else       noTone(BUZZER_PIN);      // 0：静音
    sendAck(true);
}
```

**讲解稿：**

> 这段是吃过亏之后补的安全设计。上位机曾经下发 190 度——超出了云台俯仰的物理行程，舵机一直往下顶。我们在固件端做了**双重限位**：命令入口 constrain 一次，执行函数内部再 constrain 一次，任何来路的非法角度都会被截断到 60~120 度安全区。这体现了一个原则：安全边界必须放在离硬件最近的地方，不能只信任上位机。另外我们排查发现固件原本根本没实现蜂鸣命令，上位机发指令石沉大海——补上 `@BU` 之后，整套音效反馈系统才真正落地。
